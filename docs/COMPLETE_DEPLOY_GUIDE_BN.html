প্রতিটা block আলাদাভাবে paste করে Run চাপো:

📌 SQL Block 1 — Email Tracking Table (সবার আগে এটা)

-- Email Events table + indexes + RLS create table if not exists public.email_events ( id uuid primary key default gen_random_uuid(), tracking_id text unique not null, license_key text not null, recipient text, channel text, subject text, status text default 'sent', error text, opened boolean default false, open_count int default 0, replied boolean default false, sent_at timestamptz default now(), first_opened_at timestamptz, last_opened_at timestamptz, replied_at timestamptz ); create index if not exists email_events_license_idx on public.email_events (license_key); create index if not exists email_events_tracking_idx on public.email_events (tracking_id); alter table public.email_events enable row level security;

📌 SQL Block 2 — Licenses Table

-- Licenses table create table if not exists public.licenses ( id uuid primary key default gen_random_uuid(), license_key text unique not null, customer_name text, customer_phone text, customer_email text, plan_id text, plan_name text, credits int default 0, status text default 'active', activated_at timestamptz default now(), expires_at timestamptz, created_at timestamptz default now() ); alter table public.licenses enable row level security;

📌 SQL Block 3 — Payments Table

-- Payments table create table if not exists public.payments ( id uuid primary key default gen_random_uuid(), customer_name text, customer_phone text, plan_id text, plan_name text, amount numeric, tx_id text, method text, status text default 'pending', license_key text, credits int default 0, note text, created_at timestamptz default now(), confirmed_at timestamptz );

📌 SQL Block 4 — Plans Table

-- Plans table create table if not exists public.plans ( id text primary key, name text not null, credits int default 0, price numeric default 0, duration int default 30, is_active boolean default true, sort_order int default 0, created_at timestamptz default now() ); -- Default plans insert (customize price as needed) insert into public.plans (id, name, credits, price, duration, sort_order) values ('starter', 'Starter Plan', 500, 299, 30, 1), ('pro', 'Pro Plan', 2000, 799, 30, 2), ('business', 'Business Plan', 5000, 1499, 30, 3), ('lifetime', 'Lifetime Plan', 99999, 4999, 3650, 4) on conflict (id) do nothing;

📌 SQL Block 5 — Admins Table

-- Admin table create table if not exists public.admins ( id uuid primary key default gen_random_uuid(), username text unique not null, password_hash text not null, created_at timestamptz default now() ); -- Default admin insert (password: admin123 — পরে dashboard থেকে change করো) insert into public.admins (username, password_hash) values ('admin', 'admin123') on conflict (username) do nothing;

📌 SQL Block 6 — Settings Table

-- Settings table create table if not exists public.settings ( key text primary key, value text ); insert into public.settings (key, value) values ('bkash_number', '01XXXXXXXXX'), ('nagad_number', '01XXXXXXXXX'), ('rocket_number', '01XXXXXXXXX'), ('support_wa', '01XXXXXXXXX'), ('brand_name', 'LeadOutfy Ai Pro') on conflict (key) do nothing;

📌 SQL Block 7 — Functions (PDF থেকে)

-- track_open function (with 12s prefetch filter) create or replace function public.track_open(p_tracking_id text) returns void language plpgsql security definer as $$ begin update public.email_events set opened = true, open_count = open_count + 1, first_opened_at = coalesce(first_opened_at, now()), last_opened_at = now() where tracking_id = p_tracking_id and now() - sent_at > interval '12 seconds'; end; $$;
-- email_stats function create or replace function public.email_stats(p_license_key text) returns json language sql security definer as $$ select json_build_object( 'total', count(*), 'sent', count(*) filter (where status = 'sent'), 'failed', count(*) filter (where status = 'failed'), 'opened', count(*) filter (where opened = true), 'replied', count(*) filter (where replied = true) ) from public.email_events where license_key = p_license_key; $$;
-- email_stats_buckets (today/yesterday/week/month/all) create or replace function public.email_stats_buckets(p_license_key text) returns json language sql security definer as $$ with e as ( select status, opened, replied, sent_at, (sent_at at time zone 'Asia/Dhaka')::date as d from public.email_events where license_key = p_license_key ), cur as ( select (now() at time zone 'Asia/Dhaka')::date as today ) select json_build_object( 'today', (select json_build_object('total',count(*),'sent',count(*) filter(where status='sent'), 'failed',count(*) filter(where status='failed'),'opened',count(*) filter(where opened), 'replied',count(*) filter(where replied)) from e,cur where e.d=cur.today), 'yesterday', (select json_build_object('total',count(*),'sent',count(*) filter(where status='sent'), 'failed',count(*) filter(where status='failed'),'opened',count(*) filter(where opened), 'replied',count(*) filter(where replied)) from e,cur where e.d=cur.today-1), 'week', (select json_build_object('total',count(*),'sent',count(*) filter(where status='sent'), 'failed',count(*) filter(where status='failed'),'opened',count(*) filter(where opened), 'replied',count(*) filter(where replied)) from e where sent_at>=now()-interval '7 days'), 'month', (select json_build_object('total',count(*),'sent',count(*) filter(where status='sent'), 'failed',count(*) filter(where status='failed'),'opened',count(*) filter(where opened), 'replied',count(*) filter(where replied)) from e where sent_at>=now()-interval '30 days'), 'all', (select json_build_object('total',count(*),'sent',count(*) filter(where status='sent'), 'failed',count(*) filter(where status='failed'),'opened',count(*) filter(where opened), 'replied',count(*) filter(where replied)) from e) ); $$;

8Common Problems & Fix

❌ Problem: "Invalid credentials" — Login হচ্ছে না

Fix: Supabase → Table Editor → admins table → দেখো row আছে কিনা। না থাকলে SQL Block 5 আবার run করো।

❌ Problem: Dashboard deploy হয়েছে কিন্তু blank/white screen

Fix: Vercel → তোমার project → Deployments → latest deployment-এর logs দেখো। সমস্যাটা সেখানে লেখা থাকবে। প্রায়ই এটা হয় যদি index.html সঠিকভাবে root-এ না থাকে।

❌ Problem: Data দেখাচ্ছে না / tables empty

Fix: Supabase → Table Editor → plans table-এ কিছু data insert করো। প্রথমবার plans empty থাকলে dashboard-এ কিছু দেখাবে না।

❌ Problem: Extension license verify হচ্ছে না

Fix: Extension backend (lead-outfy-backend-vercel.vercel.app) আলাদা Vercel project। সেই project-এ SUPABASE_URL এবং SUPABASE_SERVICE_KEY environment variable দেওয়া আছে কিনা check করো।

❌ Problem: Vercel deploy-এ "Build Failed"

Fix: Framework Preset → "Other" select করো। Build Command খালি রাখো। তোমার project static HTML — কোনো build লাগে না।
